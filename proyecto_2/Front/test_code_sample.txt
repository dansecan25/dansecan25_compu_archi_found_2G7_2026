# ============================================================
# CÓDIGO DE PRUEBA PARA EL SIMULADOR RISC-V
# Incluye: operaciones aritméticas, loads/stores, branches, y jal
# ============================================================

# --- Inicialización de registros ---
addi x1, x0, 5        # x1 = 5
addi x2, x0, 10       # x2 = 10
add x3, x1, x2        # x3 = x1 + x2 = 15
addi x4, x3, 2        # x4 = x3 + 2 = 17

# --- Operaciones con memoria (Store) ---
addi x10, x0, 0       # x10 = dirección base 0
sw x1, 0(x10)         # Mem[0] = 5
sw x2, 4(x10)         # Mem[4] = 10
sw x3, 8(x10)         # Mem[8] = 15
sw x4, 12(x10)        # Mem[12] = 17

# --- Operaciones con memoria (Load) ---
lw x5, 0(x10)         # x5 = Mem[0] = 5
lw x6, 4(x10)         # x6 = Mem[4] = 10
add x7, x5, x6        # x7 = x5 + x6 = 15

# --- Branch NO tomado ---
beq x1, x2, skip1     # x1 != x2, no salta
addi x8, x0, 99       # x8 = 99 (se ejecuta)
skip1:

# --- Branch SI tomado ---
beq x1, x1, skip2     # x1 == x1, salta
addi x9, x0, 88       # x9 = 88 (NO se ejecuta)
skip2:
addi x9, x0, 77       # x9 = 77 (se ejecuta)

# --- Loop simple ---
addi x11, x0, 0       # contador = 0
addi x12, x0, 3       # límite = 3
loop:
addi x11, x11, 1      # contador++
add x13, x13, x11     # acumulador += contador
beq x11, x12, endloop # si contador == 3, salir
jal x0, loop          # repetir loop
endloop:

# --- Llamada a función con JAL ---
jal x14, funcion      # llamar función, guardar retorno en x14
addi x15, x14, 1      # usar x14 después del retorno
jal x0, fin           # saltar a fin

# --- Función ---
funcion:
addi x20, x0, 100     # x20 = 100
addi x21, x20, 50     # x21 = 150
jalr x0, x14, 0       # retornar

# --- Fin del programa ---
fin:
addi x31, x0, 255     # marca de fin
nop
